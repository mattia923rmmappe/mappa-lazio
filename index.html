    <script>
        var map = L.map('map', { center: [41.70, 12.55], zoom: 10, zoomControl: false });

        var googleHybrid = L.tileLayer('https://{s}.google.com/vt/lyrs=y&x={x}&y={y}&z={z}', { maxZoom: 20, subdomains:['mt0','mt1','mt2','mt3'] }).addTo(map);
        var googleTerrain = L.tileLayer('https://{s}.google.com/vt/lyrs=p&x={x}&y={y}&z={z}', { maxZoom: 20, subdomains:['mt0','mt1','mt2','mt3'] });
        var currentBase = "sat";

        function switchBaseMap() {
            if (currentBase === "sat") {
                map.removeLayer(googleHybrid); googleTerrain.addTo(map);
                currentBase = "terrain"; document.getElementById('btn-map-type').innerHTML = "🗺️ MAPPA SATELLITE";
            } else {
                map.removeLayer(googleTerrain); googleHybrid.addTo(map);
                currentBase = "sat"; document.getElementById('btn-map-type').innerHTML = "🗺️ MAPPA TERRENO";
            }
        }

        function toggleMenu() {
            var panel = document.getElementById('main-panel');
            var btn = document.getElementById('btn-menu');
            if (panel.style.display === 'none' || panel.style.display === '') {
                panel.style.display = 'block'; btn.innerHTML = '❌ CHIUDI MENU';
            } else {
                panel.style.display = 'none'; btn.innerHTML = '⚙️ APRI MENU';
            }
        }

        function getMapsLink(lat, lng) {
            return `<a href="https://www.google.com/maps/dir/?api=1&destination=${lat},${lng}" target="_blank" class="nav-link">🚗 NAVIGA QUI</a>`;
        }

        var respectLayer = L.layerGroup().addTo(map);
        function addRespectCircle(lat, lng) {
            L.circle([lat, lng], { radius: 150, color: '#ff0000', fillColor: '#ff0000', fillOpacity: 0.15, dashArray: '5, 5', interactive: false }).addTo(respectLayer);
        }

        // --- DATABASE STORICO E MILITARE ---
        var wwiiPoints = L.layerGroup().addTo(map);
        var bunkerDB = [
            {loc: [41.7336, 12.2725], name: "Bunker Ostia Lido", cod: "OST-01"},
            {loc: [41.7450, 12.2610], name: "Difesa Idroscalo", cod: "OST-02"},
            {loc: [41.7120, 12.3250], name: "Bunker Castel Porziano", cod: "CPZ-01"},
            {loc: [41.6444, 12.6391], name: "Caposaldo Campoleone", cod: "CPL-05"},
            {loc: [41.6520, 12.6210], name: "Bunker Via della Cogna", cod: "CPL-08"},
            {loc: [41.5950, 12.6700], name: "Bunker Aprilia (The Factory)", cod: "APR-01"},
            {loc: [41.6661, 12.7052], name: "Bunker Villa Crocetta", cod: "LNV-01"},
            {loc: [41.6780, 12.6930], name: "Artiglieria Lanuvio", cod: "LNV-02"},
            {loc: [41.6700, 12.7150], name: "Postazione Tobruk", cod: "LNV-T1"},
            {loc: [41.6850, 12.7780], name: "Caposaldo Velletri", cod: "VEL-01"},
            {loc: [41.6934, 12.8125], name: "Colle S. Angela", cod: "VEL-12"},
            {loc: [41.7725, 12.9231], name: "Bunker Artena", cod: "ART-01"},
            {loc: [42.2370, 12.5117], name: "Bunker Soratte", cod: "SOR-01"},
            {loc: [41.8320, 12.4680], name: "Bunker EUR", cod: "EUR-01"},
            {loc: [41.9161, 12.4533], name: "Bunker Monte Mario", cod: "ROM-MM"},
            {loc: [41.6025, 12.5644], name: "Fosso Moletta", cod: "ARD-03"}
        ];
        bunkerDB.forEach(b => {
            L.circleMarker(b.loc, { radius: 8, color: '#000', fillColor: '#808080', fillOpacity: 1, weight: 2 })
             .bindPopup(`<b>${b.name}</b><br>CODICE: <code>${b.cod}</code><br>${getMapsLink(b.loc[0], b.loc[1])}`).addTo(wwiiPoints);
            addRespectCircle(b.loc[0], b.loc[1]);
        });

        var lineaCaesar = L.polyline([[41.53, 12.58], [41.59, 12.65], [41.65, 12.68], [41.68, 12.78], [41.72, 12.85], [41.77, 12.92]], {color: 'purple', weight: 4, dashArray: '10'}).addTo(map);

        var ancientSites = L.layerGroup().addTo(map);
        var settlements = [
            {loc: [42.0232, 12.3898], name: "Veio"}, {loc: [42.2275, 11.7891], name: "Tarquinia"},
            {loc: [41.9344, 12.0469], name: "Caere"}, {loc: [41.7247, 12.6661], name: "Alba Longa"},
            {loc: [41.8900, 12.6450], name: "Gabii"}, {loc: [41.6111, 12.5161], name: "Ardea"},
            {loc: [41.6500, 12.4833], name: "Lavinium"}, {loc: [41.8402, 12.8924], name: "Praeneste"}
        ];
        settlements.forEach(s => {
            L.marker(s.loc, { icon: L.divIcon({ html: '🏛️', iconSize: [20, 20], className:'' }) }).bindPopup(`<b>${s.name}</b><br>${getMapsLink(s.loc[0], s.loc[1])}`).addTo(ancientSites);
            addRespectCircle(s.loc[0], s.loc[1]);
        });

        var historicalBattles = L.layerGroup().addTo(map);
        var battles = [{loc: [41.961, 12.493], name: "Saxa Rubra"}, {loc: [41.722, 12.674], name: "Lago Regillo"}, {loc: [41.685, 12.778], name: "Batt. Velletri"}];
        battles.forEach(b => { L.marker(b.loc, { icon: L.divIcon({ html: '⚔️', iconSize: [20, 20], className:'' }) }).bindPopup(`<b>${b.name}</b><br>${getMapsLink(b.loc[0], b.loc[1])}`).addTo(historicalBattles); });

        // --- SISTEMA WMS REGIONALE DINAMICO ---
        var wmsDatabase = {
            'lazio': {
                url: 'https://geoportale.regione.lazio.it/geoserver/wms',
                arch: 'geonode:aree_archeologiche',
                ptpr: 'geonode:ptpr_tavole_a_beni_paesaggistici'
            },
            'toscana': {
                url: '', // INSERIRE URL WMS TOSCANA QUI
                arch: '', // INSERIRE NOME LAYER ARCHEOLOGIA TOSCANA
                ptpr: ''  // INSERIRE NOME LAYER PTPR TOSCANA
            },
            'campania': {
                url: '', // INSERIRE URL WMS CAMPANIA QUI
                arch: '',
                ptpr: '' 
            }
        };

        var layerArch = null;
        var layerPtpr = null;
        var isArchActive = true; // Partono accesi
        var isPtprActive = true; // Partono accesi

        function caricaWMS(regioneKey) {
            // Rimuove i vecchi layer se esistono
            if (layerArch && map.hasLayer(layerArch)) map.removeLayer(layerArch);
            if (layerPtpr && map.hasLayer(layerPtpr)) map.removeLayer(layerPtpr);

            var dati = wmsDatabase[regioneKey];
           
            // Se la regione ha i link inseriti, crea i layer
            if(dati.url !== '') {
                layerArch = L.tileLayer.wms(dati.url, { layers: dati.arch, format: 'image/png', transparent: true, opacity: 0.6 });
                layerPtpr = L.tileLayer.wms(dati.url, { layers: dati.ptpr, format: 'image/png', transparent: true, opacity: 0.4 });
               
                if(isArchActive) layerArch.addTo(map);
                if(isPtprActive) layerPtpr.addTo(map);
            } else {
                alert("Attenzione: I link WMS per " + regioneKey.toUpperCase() + " non sono ancora stati inseriti nel codice.");
            }
        }

        function cambiaRegione() {
            var scelt = document.getElementById("regione-selector").value;
            document.getElementById("val-regione").innerText = scelt.toUpperCase();
            caricaWMS(scelt);
        }

        // Inizializza il Lazio all'avvio
        caricaWMS('lazio');

        function toggleL(type) {
            if(type==='arch') {
                if(map.hasLayer(layerArch)) { map.removeLayer(layerArch); isArchActive = false; }
                else { map.addLayer(layerArch); isArchActive = true; }
            } else if(type==='ptpr') {
                if(map.hasLayer(layerPtpr)) { map.removeLayer(layerPtpr); isPtprActive = false; }
                else { map.addLayer(layerPtpr); isPtprActive = true; }
            }
            else {
                var ly;
                if(type==='wwii') ly=wwiiPoints; else if(type==='ancient') ly=ancientSites;
                else if(type==='battles') ly=historicalBattles; else if(type==='caesar') ly=lineaCaesar;
                else if(type==='respect') ly=respectLayer;
                if(map.hasLayer(ly)) map.removeLayer(ly); else map.addLayer(ly);
            }
        }

        // --- GPS E GEOMAN ---
        map.pm.addControls({ position: 'topleft', drawMarker: true, drawPolyline: false, drawPolygon: false, removalMode: true });
        map.on('pm:create', function(e) {
            if (e.shape === 'Marker') {
                var lat = e.layer.getLatLng().lat;
                var lng = e.layer.getLatLng().lng;
                addRespectCircle(lat, lng);
                var nome = prompt("Nome nuovo ritrovamento:");
                if(nome) e.layer.bindPopup(`<b>${nome}</b><br>${getMapsLink(lat, lng)}`).openPopup();
            }
        });

        function attivaGps() { map.locate({setView:true, maxZoom:17, enableHighAccuracy:true}); }
        map.on('locationfound', e => { L.circle(e.latlng, {radius: 10, color: '#ffff00', fillColor: '#0000ff', fillOpacity: 0.5}).addTo(map); });
        map.on('zoomend', () => document.getElementById('val-zoom').innerText = map.getZoom());
    </script>
